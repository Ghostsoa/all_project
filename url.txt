// WebDocument 表示提取的网页内容
type WebDocument struct {
	URL          string         `json:"url"`
	Title        string         `json:"title"`
	MainText     string         `json:"main_text"`
	Author       string         `json:"author,omitempty"`
	Excerpt      string         `json:"excerpt,omitempty"`
	SiteName     string         `json:"site_name,omitempty"`
	Image        string         `json:"image,omitempty"`
	Favicon      string         `json:"favicon,omitempty"`
	Chunks       []ContentChunk `json:"chunks,omitempty"`
	DocumentID   string         `json:"document_id"`
	FetchedAt    time.Time      `json:"fetched_at"`
	ContentStats ContentStats   `json:"stats"`
}

type ContentChunk struct {
	Position int    `json:"position"`
	Content  string `json:"content"`
	Length   int    `json:"length"`
}

type ContentStats struct {
	TotalLength int `json:"total_length"`
	WordCount   int `json:"word_count"`
}

// ============================================================================
// OpenAI API 相关
// ============================================================================

type OpenAIRequest struct {
	Model    string          `json:"model"`
	Messages []OpenAIMessage `json:"messages"`
	Stream   bool            `json:"stream"`
}

type OpenAIMessage struct {
	Role    string `json:"role"`
	Content string `json:"content"`
}

type OpenAIResponse struct {
	Choices []struct {
		Message OpenAIMessage `json:"message"`
	} `json:"choices"`
	Error *struct {
		Message string `json:"message"`
		Type    string `json:"type"`
	} `json:"error,omitempty"`
}

// ReadURLContent 使用 Readability 提取网页内容
func ReadURLContent(url string) (*WebDocument, error) {
	// 使用 readability 解析
	article, err := readability.FromURL(url, 30*time.Second)
	if err != nil {
		return nil, fmt.Errorf("readability 解析失败: %w", err)
	}

	// 清理文本内容，移除乱码
	cleanText := cleanInvalidChars(article.TextContent)

	// 构建文档
	doc := &WebDocument{
		URL:        url,
		Title:      cleanInvalidChars(article.Title),
		MainText:   cleanText,
		Author:     cleanInvalidChars(article.Byline),
		Excerpt:    cleanInvalidChars(article.Excerpt),
		SiteName:   cleanInvalidChars(article.SiteName),
		Image:      article.Image,
		Favicon:    article.Favicon,
		DocumentID: generateDocID(),
		FetchedAt:  time.Now(),
	}

	// 计算统计信息
	doc.ContentStats.TotalLength = len(doc.MainText)
	doc.ContentStats.WordCount = len(strings.Fields(doc.MainText))

	// 分块（每块 2000 字符）
	chunkContent(doc, 2000)

	return doc, nil
}

// cleanInvalidChars 清理无效字符和乱码
func cleanInvalidChars(text string) string {
	// 移除 Unicode 替换字符 (U+FFFD)
	text = strings.ReplaceAll(text, "\ufffd", "")

	// 移除控制字符（保留换行、制表符）
	var cleaned strings.Builder
	for _, r := range text {
		// 保留可打印字符、换行、制表符、空格
		if r >= 32 || r == '\n' || r == '\t' || r == '\r' {
			// 跳过私有区域和特殊符号
			if r < 0xE000 || r > 0xF8FF {
				cleaned.WriteRune(r)
			}
		}
	}

	return strings.TrimSpace(cleaned.String())
}

// chunkContent 将内容分块
func chunkContent(doc *WebDocument, chunkSize int) {
	content := doc.MainText
	totalLen := len(content)

	if totalLen <= chunkSize {
		doc.Chunks = []ContentChunk{
			{Position: 0, Content: content, Length: totalLen},
		}
		return
	}

	position := 0
	for i := 0; i < totalLen; i += chunkSize {
		end := i + chunkSize
		if end > totalLen {
			end = totalLen
		}

		chunk := ContentChunk{
			Position: position,
			Content:  content[i:end],
			Length:   end - i,
		}
		doc.Chunks = append(doc.Chunks, chunk)
		position++
	}
}

// generateDocID 生成文档 ID
func generateDocID() string {
	return fmt.Sprintf("doc_%d", time.Now().UnixNano())
}

// ============================================================================
// AI 摘要
// ============================================================================

func summarizeWithAI(doc *WebDocument) (string, error) {
	prompt := fmt.Sprintf(`请分析以下根据爬虫爬取的网页内容并生成详细摘要：

标题：%s
描述：%s

内容：
%s

关键点：
1、如果有关键的相关内容，如：示例代码、核心方法、知识点等，你需要完整提取并输出
2、你需要尽最大程度分析网页内容，提取主要信息
3、过滤无用的广告信息


**重要：直接输出文本内容，不要使用 JSON 或其他结构化格式。**`, doc.Title, doc.Excerpt, doc.MainText)

	// 限制内容长度
	if len(prompt) > 8000 {
		prompt = prompt[:8000] + "...\n\n(内容过长已截断)"
	}

	reqBody := OpenAIRequest{
		Model: OPENAI_MODEL,
		Messages: []OpenAIMessage{
			{Role: "user", Content: prompt},
		},
		Stream: false,
	}

	jsonData, err := json.Marshal(reqBody)
	if err != nil {
		return "", fmt.Errorf("序列化请求失败: %w", err)
	}

	req, err := http.NewRequest("POST", OPENAI_API_ENDPOINT, bytes.NewBuffer(jsonData))
	if err != nil {
		return "", fmt.Errorf("创建请求失败: %w", err)
	}

	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Bearer "+OPENAI_API_KEY)

	client := &http.Client{Timeout: 60 * time.Second}
	resp, err := client.Do(req)
	if err != nil {
		return "", fmt.Errorf("发送请求失败: %w", err)
	}
	defer resp.Body.Close()

	bodyBytes, err := io.ReadAll(resp.Body)
	if err != nil {
		return "", fmt.Errorf("读取响应失败: %w", err)
	}

	if resp.StatusCode != http.StatusOK {
		return "", fmt.Errorf("API 请求失败 (状态码 %d): %s", resp.StatusCode, string(bodyBytes))
	}

	var openaiResp OpenAIResponse
	if err := json.Unmarshal(bodyBytes, &openaiResp); err != nil {
		return "", fmt.Errorf("解析响应失败: %w\n原始响应: %s", err, string(bodyBytes))
	}

	if openaiResp.Error != nil {
		return "", fmt.Errorf("API 错误: %s", openaiResp.Error.Message)
	}

	if len(openaiResp.Choices) > 0 {
		return openaiResp.Choices[0].Message.Content, nil
	}

	return "", fmt.Errorf("未收到有效响应")
}

func main() {
	fmt.Println("============================================================")
	fmt.Println("     🌐 智能网页内容提取 + AI 摘要工具 v2.0 (Readability)")
	fmt.Println("============================================================")
	fmt.Println()

	for {
		fmt.Print("\n请输入URL (输入 'q' 或 'exit' 退出): ")

		var url string
		fmt.Scanln(&url)

		url = strings.TrimSpace(url)
		if url == "q" || url == "exit" || url == "quit" {
			fmt.Println("\n👋 再见！")
			break
		}

		if url == "" {
			fmt.Println("❌ URL 不能为空")
			continue
		}

		if !strings.HasPrefix(url, "http://") && !strings.HasPrefix(url, "https://") {
			fmt.Println("❌ URL 必须以 http:// 或 https:// 开头")
			continue
		}

		fmt.Println()
		fmt.Printf("🔍 正在获取网页内容: %s\n", url)
		fmt.Println(strings.Repeat("=", 80))

		// 提取内容
		doc, err := ReadURLContent(url)
		if err != nil {
			fmt.Printf("❌ 错误: %v\n", err)
			continue
		}

		// 输出基本信息
		fmt.Printf("\n📄 标题: %s\n", doc.Title)
		if doc.Author != "" {
			fmt.Printf("👤 作者: %s\n", doc.Author)
		}
		if doc.Excerpt != "" {
			fmt.Printf("📝 摘录: %s\n", doc.Excerpt)
		}
		if doc.SiteName != "" {
			fmt.Printf("🌐 网站: %s\n", doc.SiteName)
		}

		fmt.Printf("\n📊 统计信息:\n")
		fmt.Printf("  - 总字符数: %d\n", doc.ContentStats.TotalLength)
		fmt.Printf("  - 单词数: %d\n", doc.ContentStats.WordCount)
		fmt.Printf("  - 内容块数: %d\n", len(doc.Chunks))

		// 调用大模型生成摘要
		fmt.Printf("\n🤖 正在调用 AI 大模型生成摘要...\n")
		summary, err := summarizeWithAI(doc)
		if err != nil {
			fmt.Printf("⚠️  AI 摘要失败: %v\n", err)
			fmt.Println("\n📖 原始内容预览:")
			fmt.Println(strings.Repeat("-", 80))
			if len(doc.MainText) > 500 {
				fmt.Println(doc.MainText[:500] + "...")
			} else {
				fmt.Println(doc.MainText)
			}
			fmt.Println(strings.Repeat("-", 80))
		} else {
			fmt.Println("\n✨ AI 智能摘要:")
			fmt.Println(strings.Repeat("=", 80))
			fmt.Println(summary)
			fmt.Println(strings.Repeat("=", 80))
		}

		// 询问是否保存 JSON
		fmt.Print("\n💾 是否保存完整数据为 JSON? (y/n): ")
		var input string
		fmt.Scanln(&input)

		if strings.ToLower(strings.TrimSpace(input)) == "y" {
			filename := fmt.Sprintf("content_%s.json", doc.DocumentID)
			data, _ := json.MarshalIndent(doc, "", "  ")
			if err := os.WriteFile(filename, data, 0644); err != nil {
				fmt.Printf("❌ 保存失败: %v\n", err)
			} else {
				fmt.Printf("✅ 已保存到: %s\n", filename)
			}
		}

		fmt.Println("\n" + strings.Repeat("=", 80))
	}
}
